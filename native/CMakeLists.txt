cmake_minimum_required(VERSION 3.24)
project(dome-connect-dialog LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(FLTK CONFIG REQUIRED HINTS /opt/fltk-vr/toolkit/share/fltk)
find_package(nlohmann_json REQUIRED)
find_package(ZLIB REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
add_executable(dome-connect-dialog dialog.cpp)
target_include_directories(dome-connect-dialog PRIVATE ../../flterm-xr/xr)
target_link_libraries(dome-connect-dialog PRIVATE fltk::fltk-shared nlohmann_json::nlohmann_json PkgConfig::GIO ZLIB::ZLIB)
set_target_properties(dome-connect-dialog PROPERTIES INSTALL_RPATH /opt/fltk-vr/toolkit/lib)
install(TARGETS dome-connect-dialog DESTINATION share/kdeconnect-dome)
